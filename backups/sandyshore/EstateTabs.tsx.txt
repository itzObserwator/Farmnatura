"use client";
import { useRef, useState } from "react";
import gsap from "gsap";
import { ButtonLink, Photo } from "@/components/ui/Primitives";
const themes = [
  {
    label: "Farmland",
    title: "Your own\nliving land",
    text: "Discover 110+ acres of managed farmland in Kandukur, near Hyderabad. Own your land, grow with the seasons, and let our team care for everyday farming.",
    image: "orchard-retina.jpg",
    alt: "Farm Natura’s orchard trees and cultivated soil",
    href: "/farm-lands-for-sale-in-hyderabad",
  },
  {
    label: "Natural farming",
    title: "Rooted in\nnatural farming",
    text: "Years of soil care, indigenous seeds, and seasonal cultivation. A relationship with the earth that values its health and the generations who follow.",
    image: "farm-care-retina.jpg",
    alt: "Farm Natura’s team caring for planted fields",
    href: "/natural-farming",
  },
  {
    label: "Farm living",
    title: "A place to\nfeel at home",
    text: "A shaded veranda. A walk through the fields. Time with your family, without a timetable. Make room for a slower rhythm at Farm Natura.",
    image: "pavilion-retina.jpg",
    alt: "Farm Natura’s open-sided pavilion",
    href: "/farmhouses-for-sale-in-hyderabad",
  },
];
export default function EstateTabs() {
  const [active, setActive] = useState(0);
  const panel = useRef<HTMLDivElement>(null);
  const pending = useRef(false);
  const buttons = useRef<(HTMLButtonElement | null)[]>([]);
  function select(index: number) {
    if (index === active || pending.current) return;
    const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    if (reduce) {
      setActive(index);
      return;
    }
    pending.current = true;
    gsap.to(panel.current, {
      opacity: 0,
      x: index > active ? -30 : 30,
      duration: 0.3,
      ease: "power2.in",
      onComplete: () => {
        setActive(index);
        requestAnimationFrame(() =>
          gsap.fromTo(
            panel.current,
            { opacity: 0, x: index > active ? 40 : -40 },
            {
              opacity: 1,
              x: 0,
              duration: 0.7,
              ease: "power4.out",
              clearProps: "all",
              onComplete: () => {
                pending.current = false;
              },
            },
          ),
        );
      },
    });
  }
  const theme = themes[active];
  return (
    <section className="estate-tabs" aria-label="Explore Farm Natura">
      <h2 className="visually-hidden">Explore the estate, farming, and farm living</h2>
      <div className="estate-tab-nav" role="tablist" aria-label="Farm Natura experiences">
        {themes.map((item, i) => (
          <button
            key={item.label}
            ref={(node) => {
              buttons.current[i] = node;
            }}
            id={`estate-tab-${i}`}
            role="tab"
            aria-selected={active === i}
            aria-controls="estate-panel"
            tabIndex={active === i ? 0 : -1}
            onClick={() => select(i)}
            onKeyDown={(e) => {
              let next = i;
              if (e.key === "ArrowRight") next = (i + 1) % 3;
              else if (e.key === "ArrowLeft") next = (i + 2) % 3;
              else if (e.key === "Home") next = 0;
              else if (e.key === "End") next = 2;
              else return;
              e.preventDefault();
              buttons.current[next]?.focus();
              select(next);
            }}
          >
            <span className="tab-diamond" aria-hidden="true">
              ✦
            </span>
            {item.label}
          </button>
        ))}
      </div>
      <div
        ref={panel}
        className="estate-panel"
        role="tabpanel"
        id="estate-panel"
        aria-labelledby={`estate-tab-${active}`}
        tabIndex={0}
      >
        <div className="estate-panel-copy">
          <h3>
            {theme.title.split("\n").map((line, i) => (
              <span key={i}>{line}</span>
            ))}
          </h3>
          <p>{theme.text}</p>
          <ButtonLink href={theme.href}>Learn more</ButtonLink>
          <span className="panel-number">0{active + 1} / 03</span>
        </div>
        <Photo src={theme.image} alt={theme.alt} />
      </div>
    </section>
  );
}
