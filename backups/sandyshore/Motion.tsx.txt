"use client";
import { useEffect, useRef } from "react";
import { usePathname, useRouter } from "next/navigation";
import gsap from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";
import Lenis from "lenis";

/** Shared motion: route curtain, scrolling, parallax, progressive letters, and frame progress. */
export default function Motion() {
  const pathname = usePathname();
  const router = useRouter();
  const curtain = useRef<HTMLDivElement>(null);
  const navigating = useRef(false);
  useEffect(() => {
    if (curtain.current) {
      gsap.to(curtain.current, {
        scaleY: 0,
        transformOrigin: "50% 0%",
        duration: 0.75,
        ease: "power4.inOut",
        onComplete: () => {
          navigating.current = false;
        },
      });
    }
    const click = (event: MouseEvent) => {
      if (
        event.defaultPrevented ||
        event.button !== 0 ||
        event.metaKey ||
        event.ctrlKey ||
        event.shiftKey ||
        event.altKey ||
        window.matchMedia("(prefers-reduced-motion: reduce)").matches
      )
        return;
      const anchor = (event.target as HTMLElement).closest<HTMLAnchorElement>("a[href]");
      if (!anchor || anchor.target === "_blank" || anchor.hasAttribute("download")) return;
      const url = new URL(anchor.href, window.location.href);
      if (url.origin !== window.location.origin || url.pathname === pathname || url.hash) return;
      event.preventDefault();
      event.stopPropagation();
      if (navigating.current) return;
      navigating.current = true;
      gsap.fromTo(
        curtain.current,
        { scaleY: 0, transformOrigin: "50% 100%" },
        {
          scaleY: 1,
          duration: 0.55,
          ease: "power4.inOut",
          onComplete: () => {
            router.push(url.pathname + url.search);
          },
        },
      );
    };
    document.addEventListener("click", click, true);
    return () => document.removeEventListener("click", click, true);
  }, [pathname, router]);
  useEffect(() => {
    gsap.registerPlugin(ScrollTrigger);
    const media = gsap.matchMedia();
    let lenis: Lenis | undefined;
    let tick: ((time: number) => void) | undefined;
    const frame = gsap.to(".rail-fill", {
      scaleY: 1,
      ease: "none",
      scrollTrigger: {
        trigger: document.documentElement,
        start: "top top",
        end: "bottom bottom",
        scrub: true,
      },
    });
    media.add("(prefers-reduced-motion: no-preference)", () => {
      const entrances = document.querySelectorAll(".hero-enter");
      if (entrances.length)
        gsap.from(entrances, {
          opacity: 0,
          y: 24,
          filter: "blur(5px)",
          duration: 1,
          stagger: 0.1,
          ease: "power3.out",
          clearProps: "all",
        });
      gsap.utils.toArray<HTMLElement>("[data-reveal]").forEach((element) =>
        gsap.from(element, {
          opacity: 0,
          y: 35,
          duration: 1,
          ease: "power3.out",
          scrollTrigger: { trigger: element, start: "top 92%", once: true },
          clearProps: "all",
        }),
      );
      gsap.utils.toArray<HTMLElement>(".value-heading").forEach((heading, index) => {
        const story = document.querySelector(`[data-value-story="${index}"]`);
        gsap.fromTo(
          heading.querySelectorAll(".value-letter"),
          { color: "#dfd4cf" },
          {
            color: "#181715",
            stagger: 0.06,
            ease: "none",
            scrollTrigger: { trigger: story, start: "top 75%", end: "top 20%", scrub: true },
          },
        );
      });
      gsap.utils.toArray<HTMLElement>(".botanical-cutout").forEach((element, index) =>
        gsap.fromTo(
          element,
          { y: 40, rotation: (index % 2 ? 1 : -1) * 10 },
          {
            y: -40,
            rotation: (index % 2 ? -1 : 1) * 10,
            ease: "none",
            scrollTrigger: {
              trigger: ".reference-portrait",
              start: "top bottom",
              end: "bottom top",
              scrub: true,
            },
          },
        ),
      );
      if (window.matchMedia("(min-width: 900px) and (pointer: fine)").matches) {
        lenis = new Lenis({
          duration: 1.1,
          smoothWheel: true,
          anchors: true,
          prevent: (node) => !!node.closest("dialog,.home-story"),
        });
        lenis.on("scroll", ScrollTrigger.update);
        tick = (time) => lenis?.raf(time * 1000);
        gsap.ticker.add(tick);
        gsap.utils.toArray<HTMLElement>("[data-parallax] img").forEach((element) =>
          gsap.fromTo(
            element,
            { yPercent: -5, scale: 1.12 },
            {
              yPercent: 5,
              ease: "none",
              scrollTrigger: {
                trigger: element.parentElement,
                start: "top bottom",
                end: "bottom top",
                scrub: true,
              },
            },
          ),
        );
      }
      return () => {
        if (tick) gsap.ticker.remove(tick);
        lenis?.destroy();
      };
    });
    const yellow = document.querySelector(".visit-banner");
    const theme = yellow
      ? ScrollTrigger.create({
          trigger: yellow,
          start: "top 60px",
          end: "max",
          toggleClass: { targets: document.body, className: "theme-yellow" },
        })
      : null;
    const refresh = () => ScrollTrigger.refresh();
    document.fonts.ready.then(refresh);
    window.addEventListener("load", refresh);
    return () => {
      frame.scrollTrigger?.kill();
      frame.kill();
      media.revert();
      theme?.kill();
      document.body.classList.remove("theme-yellow");
      window.removeEventListener("load", refresh);
    };
  }, [pathname]);
  return (
    <div className="route-curtain" ref={curtain} aria-hidden="true">
      <span>farmnatura</span>
    </div>
  );
}
